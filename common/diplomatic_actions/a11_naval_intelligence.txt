naval_intelligence = {
	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		battlefleet_tactics
	}

	selectable = {
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
	}

	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
		institution_investment_level = {
			institution = institution_espionage
			value >= 3
		}
	}

	possible = {
		is_revolutionary = no
		in_default = no
		NOT = { has_war_with = scope:target_country }
		NOT = { has_variable = recently_ran_naval_intel }
		scope:target_country = {
			navy_size >= 5
		}
		has_diplomatic_relevance = scope:target_country
	}

	pact = {
		cost = 200

		is_two_sided_pact = no
		show_in_outliner = yes

		monthly_effect = {
			custom_tooltip = {
				text = naval_intelligence_monthly_effect
				if = {
					limit = {
						NOT = { has_variable = ongoing_naval_intel }
					}
					trigger_event = { id = adv_dip_event.140 }
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				NOT = {
			OR = {
				has_war_with = scope:target_country
				is_country_type = decentralized
				scope:target_country = { is_country_type = decentralized }
				in_default = yes
				has_variable = recently_ran_naval_intel
			}
				}
			}
		}
	}

	ai = {
		max_influence_spending_fraction = 0.15

		propose_score = {
			value = 5
			if = {
				limit = {
					scope:target_country = {
						navy_size >= 20
					}
				}
				add = 20
			}
		}

		will_propose = {
			scaled_debt <= 0
			is_at_war = no
			scope:target_country = {
				relations:root <= relations_threshold:cold
			}
		}

		will_break = {
			OR = {
				scaled_debt > 0.25
				AND = {
					scaled_debt > 0
					is_at_war = yes
				}
			}
		}
	}
}
