naval_intelligence = {
	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		battlefleet_tactics
	}

	selectable = {
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
	}

	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
		OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
		institution_investment_level = {
			institution = institution_espionage
			value >= 3
		}
	}

	possible = {
		is_revolutionary = no
		in_default = no
		NOT = { has_war_with = scope:target_country }
		NOT = { has_variable = recently_ran_naval_intel }
		scope:target_country = {
			navy_size >= 5
		}
		has_diplomatic_relevance = scope:target_country
		adv_dip_operation_launch_gate = yes
	}



	accept_effect = {
		adv_dip_begin_operation = { CODE = 11 }
	}

	ai = {
		max_influence_spending_fraction = 0.15

		propose_score = {
			value = 5
			if = {
				limit = {
					scope:target_country = {
						navy_size >= 20
					}
				}
				add = 20
			}
		}

		will_propose = {
			scaled_debt <= 0
			is_at_war = no
			scope:target_country = {
				relations:root <= relations_threshold:cold
			}
		}


	}
}
