steal_commander = {
    groups = {
        general
    }
    requires_approval = no
    should_notify_third_parties = no

    unlocking_technologies = {
        standing_army
    }

    selectable = {
        NOT = { is_country_type = decentralized }
        is_revolutionary = no
    }
    
    potential = {
        NOT = { is_country_type = decentralized }
        scope:target_country = {
            NOT = { is_country_type = decentralized }
            any_scope_character = {
                OR = {
                    has_role_of_type = general
                    has_role_of_type = admiral
                }
            }
        }
		OR = {
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
        institution_investment_level = {
            institution = institution_espionage
            value >= 2
        }
    }
    
    possible = {
		NOT = {
			is_target_in_variable_list = {
				name = adv_dip_op_steal_commander
				target = scope:target_country
			}
		}
        is_revolutionary = no
        in_default = no
        NOT = { has_war_with = scope:target_country }
        scope:target_country = {
            country_rank < rank_value:great_power
        }
        has_diplomatic_relevance = scope:target_country
    }
    

    
	accept_effect = {
		scope:target_country = {
			save_scope_as = spy_target_country
		}
		adv_dip_start_operation = { ACTION = steal_commander }
		add_modifier = {
			name = spy_network_cost_modifier
			multiplier = spy_cost_multiplier_small
			months = 13
		}
	}

    ai = {
        max_influence_spending_fraction = 0.15
    
        propose_score = {
            value = 10
            if = {
                limit = {
                    scope:target_country = {
                        any_scope_character = {
                            OR = {
                                has_trait = expert_offensive_planner
                                has_trait = expert_artillery_commander
                                has_trait = expert_convoy_raider
                                has_trait = expert_naval_commander
                                has_trait = expert_defensive_strategist
                            }
                        }
                    }
                }
                add = 100
            }
        }
    
        will_propose = { 
            scaled_debt <= 0
            net_fixed_income >= 25000
        }
        
            
    }
}
