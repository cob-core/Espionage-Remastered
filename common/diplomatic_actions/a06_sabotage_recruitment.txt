	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		nationalism
	}
	
	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
		institution_investment_level = {
			institution = institution_espionage
			value >= 3
		}
	}
	
	possible = {
		NOT = {
			is_target_in_variable_list = {
				name = adv_dip_op_sabotage_recruitment
				target = scope:target_country
			}
		}
		is_revolutionary = no
		in_default = no
		NOT = { has_war_with = scope:target_country }
		has_diplomatic_relevance = scope:target_country
	}
	

	
	accept_effect = {
		scope:target_country = {
			save_scope_as = spy_target_country
		}
		adv_dip_start_operation = { ACTION = sabotage_recruitment }
		add_modifier = {
			name = spy_network_cost_modifier
			multiplier = spy_cost_multiplier_small
			months = 13
		}
	}

	ai = {

		propose_score = {
			value = 10
		}
	
		will_propose = { 
			scaled_debt <= 0
			is_at_war = no
			scope:target_country = {
				has_diplomatic_pact = {
					who = root
					type = rivalry
				}
				relations:root <= relations_threshold:cold									
			}
			country_rank >= rank_value:great_power
		}
		
			
	}
}
