fund_black_market_slavery = {
	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		admiralty
	}

	potential = {
		NOT = { is_country_type = decentralized }
		has_game_rule = enable_black_market_slavery
		# We are the buyer: the contraband ships sail to OUR markets.
		OR = {
			has_law = law_type:law_slave_trade
			has_law = law_type:law_legacy_slavery
			has_law = law_type:law_debt_slavery
		}
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
		OR = {
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
		institution_investment_level = {
			institution = institution_espionage
			value >= 2
		}
	}

	possible = {
		NOT = {
			is_target_in_variable_list = {
				name = adv_dip_op_fund_black_market_slavery
				target = scope:target_country
			}
		}
		is_revolutionary = no
		in_default = no
		NOT = { has_war_with = scope:target_country }
		NOT = { has_variable = recently_black_market }
		has_diplomatic_relevance = scope:target_country
		adv_dip_operation_launch_gate = yes
	}



	accept_effect = {
		adv_dip_begin_operation = { CODE = 9 }
	}

	ai = {
		max_influence_spending_fraction = 0.10

		propose_score = {
			value = 3
			if = {
				limit = {
					scope:target_country = {
						relations:root >= relations_threshold:friendly
					}
				}
				add = 10
			}
		}

		will_propose = {
			scaled_debt <= 0
			gold_reserves > 50000
			NOT = { has_law = law_type:law_slavery_banned }
		}


	}
}
