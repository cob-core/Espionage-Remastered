fund_black_market_slavery = {
	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		admiralty
	}

	selectable = {
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
	}

	potential = {
		NOT = { is_country_type = decentralized }
		has_game_rule = enable_black_market_slavery
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			OR = {
				has_law = law_type:law_slave_trade
				has_law = law_type:law_legacy_slavery
				has_law = law_type:law_debt_slavery
			}
		}
		institution_investment_level = {
			institution = institution_espionage
			value >= 2
		}
	}

	possible = {
		is_revolutionary = no
		in_default = no
		NOT = { has_war_with = scope:target_country }
		NOT = { has_variable = recently_black_market }
		has_diplomatic_relevance = scope:target_country
	}

	pact = {
		cost = 150

		is_two_sided_pact = no
		show_in_outliner = yes

		second_modifier = {
			state_slave_import_mult = 1.0
		}

		monthly_effect = {
			custom_tooltip = {
				text = fund_black_market_slavery_monthly_effect
				random_list = {
					800 = {}
					100 = {
						scope:target_country = { save_scope_as = spy_target_country }
						root = {
							trigger_event = {
								id = adv_dip_event.155
							}
						}
					}
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				NOT = {
					OR = {
						is_country_type = decentralized
						scope:target_country = { is_country_type = decentralized }
						in_default = yes
						has_war_with = scope:target_country
						NOT = {
							scope:target_country = {
								OR = {
									has_law = law_type:law_slave_trade
									has_law = law_type:law_legacy_slavery
									has_law = law_type:law_debt_slavery
								}
							}
						}
					}
				}
			}
		}
	}

	ai = {
		max_influence_spending_fraction = 0.10

		propose_score = {
			value = 3
			if = {
				limit = {
					scope:target_country = {
						relations:root >= relations_threshold:friendly
					}
				}
				add = 10
			}
		}

		will_propose = {
			scaled_debt <= 0
			gold_reserves > 50000
			NOT = { has_law = law_type:law_slavery_banned }
		}

		will_break = {
			OR = {
				scaled_debt > 0.25
				in_default = yes
			}
		}
	}
}
