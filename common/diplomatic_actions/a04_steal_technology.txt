steal_technology = {
    groups = {
        general
    }
    requires_approval = no
    should_notify_third_parties = no

    unlocking_technologies = {
        rationalism
    }

    selectable = {
        NOT = { is_country_type = decentralized }
        is_revolutionary = no
    }
    
    potential = {
        NOT = { is_country_type = decentralized }
        scope:target_country = {
            NOT = { is_country_type = decentralized }
        }
        institution_investment_level = {
            institution = institution_espionage
            value >= 1
        }
        scope:target_country = {
			country_rank > rank_value:minor_power
		}
        OR = {
            has_game_rule = enable_tech_stealing
            AND = {
                has_game_rule = players_only_tech_stealing
                is_player = yes
            }
        }
    }
    
    possible = {
		NOT = {
			is_target_in_variable_list = {
				name = adv_dip_op_steal_technology
				target = scope:target_country
			}
		}
        is_revolutionary = no
        in_default = no
        NOT = { has_war_with = scope:target_country }
        has_diplomatic_relevance = scope:target_country
    }
    

    
	accept_effect = {
		scope:target_country = {
			save_scope_as = spy_target_country
		}
		adv_dip_start_operation = { ACTION = steal_technology }
		add_modifier = {
			name = spy_network_cost_modifier
			multiplier = spy_cost_multiplier_small
			months = 13
		}
	}

    ai = {
        max_influence_spending_fraction = 0.15
    
        propose_score = {
            value = 100
        }
    
        will_propose = {
            scaled_debt <= 0
            scope:target_country = {
				has_diplomatic_pact = {
					who = root
					type = rivalry
				}
				relations:root <= relations_threshold:cold
			}
        }
        
            
    }
}
