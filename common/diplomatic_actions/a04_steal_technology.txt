steal_technology = {
    groups = {
        general
    }
    requires_approval = no
    should_notify_third_parties = no

    unlocking_technologies = {
        rationalism
    }

    selectable = {
        NOT = { is_country_type = decentralized }
        is_revolutionary = no
    }
    
    potential = {
        NOT = { is_country_type = decentralized }
        scope:target_country = {
            NOT = { is_country_type = decentralized }
        }
        institution_investment_level = {
            institution = institution_espionage
            value >= 1
        }
        scope:target_country = {
			country_rank > rank_value:minor_power
		}
        OR = {
            has_game_rule = enable_tech_stealing
            AND = {
                has_game_rule = players_only_tech_stealing
                is_player = yes
            }
        }
    }
    
    possible = {
        is_revolutionary = no
        in_default = no
        NOT = { has_war_with = scope:target_country }
		scope:target_country.techs_researched > root.techs_researched
        has_diplomatic_relevance = scope:target_country
    }
    
    pact = {
        cost = 150
    
        is_two_sided_pact = no
        show_in_outliner = yes
        can_be_used_in_sway_offers = no
        sway_maneuvers_cost = 15
        
        first_modifier = {
        }

        second_modifier = {
        }

        monthly_effect = {
            custom_tooltip = {
                text = steal_technology_monthly_effect
                random_list = {
                    1600 = {}
                    100 = {
                    	modifier = {
                    		trigger = { institution_investment_level:institution_espionage >= target_espionage_plus_1 }
                    		factor = 1.15
                    	}
                    	modifier = {
                    		trigger = { institution_investment_level:institution_espionage <= target_espionage_minus_1 }
                    		factor = 0.85
                    	}
                    	modifier = {
                    		trigger = { institution_investment_level:institution_espionage >= target_espionage_plus_2 }
                    		factor = 1.15
                    	}
                    	modifier = {
                    		trigger = { institution_investment_level:institution_espionage <= target_espionage_minus_2 }
                    		factor = 0.85
                    	}
                    	modifier = {
                    		trigger = { institution_investment_level:institution_espionage >= target_espionage_plus_3 }
                    		factor = 1.15
                    	}
                    	modifier = {
                    		trigger = { institution_investment_level:institution_espionage <= target_espionage_minus_3 }
                    		factor = 0.85
                    	}

                        if = {
							limit = {
								scope:target_country = {
									var:turned_steal_technology_by ?= root
								}
							}
							root = {
								trigger_event = {
									id = adv_dip_event.163
								}
							}
						}
						else = {
							root = {
								trigger_event = {
									id = adv_dip_event.91
								}
							}
						}
                    }
                }
            }
        }
    
        requirement_to_maintain = {
			trigger = {
				NOT = {
            OR = {
                scope:target_country.techs_researched <= root.techs_researched
                has_war_with = scope:target_country
                is_country_type = decentralized
                scope:target_country = { is_country_type = decentralized }
                in_default = yes
                NOT = {
                    scope:target_country = {
                        any_technology = {
                            is_researched = yes
                            NOT = { root.has_technology_researched = this }
                        }
                    }
                }
                scope:target_country.techs_researched > root.techs_researched
            }
				}
			}
		}
    }
    
    ai = {
        max_influence_spending_fraction = 0.15
    
        propose_score = {
            value = 100
        }
    
        will_propose = {
            scaled_debt <= 0
            scope:target_country = {
				has_diplomatic_pact = {
					who = root
					type = rivalry
				}
				relations:root <= relations_threshold:cold
			}
        }
        
        will_break = { 
            OR = {        
                scaled_debt > 0.25
            }
        }            
    }
}
