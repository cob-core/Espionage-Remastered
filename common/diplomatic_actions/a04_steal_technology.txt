steal_technology = {
    groups = {
        general
    }
    requires_approval = no
    should_notify_third_parties = no

    unlocking_technologies = {
        rationalism
    }

    selectable = {
        NOT = { is_country_type = decentralized }
        is_revolutionary = no
    }
    
    potential = {
        NOT = { is_country_type = decentralized }
        scope:target_country = {
            NOT = { is_country_type = decentralized }
        }
		OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
        institution_investment_level = {
            institution = institution_espionage
            value >= 1
        }
        scope:target_country = {
			country_rank > rank_value:minor_power
		}
        OR = {
            has_game_rule = enable_tech_stealing
            AND = {
                has_game_rule = players_only_tech_stealing
                is_player = yes
            }
        }
    }
    
    possible = {
		NOT = {
			is_target_in_variable_list = {
				name = adv_dip_op_steal_technology
				target = scope:target_country
			}
		}
        is_revolutionary = no
        in_default = no
        NOT = { has_war_with = scope:target_country }
        has_diplomatic_relevance = scope:target_country
        adv_dip_operation_launch_gate = yes
    }
    

    
	accept_effect = {
		adv_dip_begin_operation = { CODE = 4 }
	}

    ai = {
        max_influence_spending_fraction = 0.15
    
        propose_score = {
            value = 100
        }
    
        will_propose = {
            scaled_debt <= 0
            scope:target_country = {
				OR = {
					has_diplomatic_pact = {
						who = root
						type = rivalry
					}
					relations:root <= relations_threshold:hostile
				}

            }
        }
        
            
    }
}
