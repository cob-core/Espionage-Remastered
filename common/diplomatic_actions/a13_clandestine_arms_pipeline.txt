clandestine_arms_pipeline = {
	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		international_relations
	}

	selectable = {
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
	}

	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
		OR = {
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
		institution_investment_level = {
			institution = institution_espionage
			value >= 2
		}
	}

	possible = {
		NOT = {
			is_target_in_variable_list = {
				name = adv_dip_op_clandestine_arms_pipeline
				target = scope:target_country
			}
		}
		is_revolutionary = no
		in_default = no
		is_at_war = no
		NOT = { has_variable = recently_pipeline }
		scope:target_country = {
			is_at_war = yes
		}
		has_diplomatic_relevance = scope:target_country
		adv_dip_operation_launch_gate = yes
	}



	accept_effect = {
		adv_dip_begin_operation = { CODE = 8 }
	}

	ai = {
		max_influence_spending_fraction = 0.10

		propose_score = {
			value = 5
			if = {
				limit = {
					scope:target_country = {
						relations:root >= relations_threshold:friendly
					}
				}
				add = 15
			}
		}

		will_propose = {
			scaled_debt <= 0
			gold_reserves > 50000
		}


	}
}
