assassinate_political_leader = {
	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		general_staff
	}
	
	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
		NOT = {  has_variable = active_assassination }	
		has_technology_researched = colonization
		OR = {
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
		institution_investment_level = {
			institution = institution_espionage
			value >= 5
		}
	}
	
	possible = {
		NOT = { has_variable = ongoing_plans }
		any_scope_character = {
			has_role = character_role_operative
			is_busy = no
			NOT = { has_variable = adv_dip_captured }
		}
		is_revolutionary = no
		in_default = no
		has_diplomatic_relevance = scope:target_country
		scope:target_country = {
			NOT = {
				OR = {
					has_modifier = recent_assassination_modifier
				}
			}
		}
	}


	
	accept_effect = {
		scope:target_country = {
			save_scope_as = spy_target_country
		}
		set_variable = ongoing_plans
		add_modifier = {
			name = spy_network_cost_modifier
			multiplier = spy_cost_multiplier_large
			months = 25
		}
		trigger_event = { id = adv_dip_assassination_event.100 }
	}

	
	ai = {

		propose_score = {
			value = 5
		}
	
		will_propose = { 
			scaled_debt <= 0
			is_at_war = no
			scope:target_country = {
				OR = {
					has_diplomatic_pact = {
						who = root
						type = rivalry
					}
					relations:root <= relations_threshold:hostile
				}

			}
			country_rank >= rank_value:great_power
		}
		
		will_break = { 
			OR = {		
				scaled_debt > 0.25
				AND = {
					scaled_debt > 0
					is_at_war = yes
				}				
			}					
		}			
	}
}
