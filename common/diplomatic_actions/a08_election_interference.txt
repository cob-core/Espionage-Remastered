election_interference = {
	groups = {
		general
	}
	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		mass_communication
	}
	
	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			OR = {
				scope:target_country = {
					any_interest_group = {
						has_party = yes
					}
				}
			}
		}
		OR = {
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
	institution_investment_level = {
		institution = institution_espionage
		value >= 4
	}
}
	
	possible = {
		NOT = {
			is_target_in_variable_list = {
				name = adv_dip_op_election_interference
				target = scope:target_country
			}
		}
		is_revolutionary = no
		in_default = no
		scope:target_country = {
			any_interest_group = {
				has_party = yes
			}
		}
	}
	

	
	accept_effect = {
		scope:target_country = {
			save_scope_as = spy_target_country
		}
		adv_dip_start_operation = { ACTION = election_interference }
		add_modifier = {
			name = spy_network_cost_modifier
			multiplier = spy_cost_multiplier_medium
			months = 13
		}
	}

	ai = {

		propose_score = {
			value = 5
		}
	
		will_propose = { 
			scaled_debt <= 0
			is_at_war = no
			scope:target_country = {
				OR = {
					has_diplomatic_pact = {
						who = root
						type = rivalry
					}
					relations:root <= relations_threshold:hostile
				}

			}
			country_rank >= rank_value:major_power
		}
		
			
	}
}
