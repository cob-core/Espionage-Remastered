adv_dip_sgui_recruit_operative = {
	scope = country

	is_shown = {
		OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
	}

	is_valid = {
		custom_tooltip = {
			text = adv_dip_recruit_cap_tt
			OR = {
				NOT = { has_variable_list = adv_dip_operatives }
				NOT = {
					list_size = {
						name = adv_dip_operatives
						value >= institution_investment_level:institution_espionage
					}
				}
			}
		}
	}

	effect = {
		custom_tooltip = adv_dip_recruit_tt
		adv_dip_covert_charge = { COST = adv_dip_recruit_cost WEEKS = 26 }
		create_character = {
			role = character_role_operative
			culture = primary_culture
			on_created = {
				save_scope_as = new_operative
				random_list = {
					1 = { add_trait = cryptographer }
					1 = { add_trait = master_of_disguise }
					1 = { add_trait = saboteur }
					1 = { add_trait = silver_tongue }
				}
				owner = {
					add_to_variable_list = {
						name = adv_dip_operatives
						target = prev
					}
				}
			}
		}
	}
}


# Fires an idle operative: struck from the roster and stripped of the role, so
# the engine retires the character naturally. Agents in the field must be
# recalled (operation cancelled) before they can be dismissed.
adv_dip_sgui_dismiss_operative = {
	scope = country
	saved_scopes = { candidate }

	is_shown = {
		always = yes
	}

	is_valid = {
		scope:candidate ?= {
			is_busy = no
			has_role = character_role_operative
		}
	}

	effect = {
		custom_tooltip = adv_dip_dismiss_operative_tt
		if = {
			limit = { exists = scope:candidate }
			remove_list_variable = {
				name = adv_dip_operatives
				target = scope:candidate
			}
			scope:candidate = {
				save_scope_as = adv_dip_spurned_operative
				interest_group ?= {
					add_modifier = {
						name = adv_dip_spurned_service_modifier
						months = 12
					}
				}
				remove_variable = adv_dip_captured
				remove_character_role = character_role_operative
			}
			# A lesser echo of a fired Director: one in ten shops their
			# knowledge to a hostile power.
			random_list = {
				90 = {}
				10 = {
					if = {
						limit = {
							any_country = {
								NOT = { this = root }
								OR = {
									has_diplomatic_pact = {
										who = root
										type = rivalry
									}
									relations:root <= relations_threshold:hostile
								}
							}
						}
						random_country = {
							limit = {
								NOT = { this = root }
								OR = {
									has_diplomatic_pact = {
										who = root
										type = rivalry
									}
									relations:root <= relations_threshold:hostile
								}
							}
							save_scope_as = adv_dip_captor_country
						}
						trigger_event = { id = adv_dip_event.243 }
					}
				}
			}
		}
	}
}

# Backs one party in a pending election-interference decision. Root = us,
# scope:target = the country whose election it is, scope:party = the choice.
adv_dip_sgui_back_party = {
	scope = country
	saved_scopes = { target party }

	is_shown = {
		always = yes
	}

	is_valid = {
		is_target_in_variable_list = {
			name = adv_dip_election_choice
			target = scope:target
		}
		scope:target ?= {
			in_election_campaign = yes
		}
	}

	effect = {
		custom_tooltip = adv_dip_back_party_tt
		if = {
			limit = {
				exists = scope:target
				exists = scope:party
			}
			remove_list_variable = {
				name = adv_dip_election_choice
				target = scope:target
			}
			scope:party = {
				set_variable = {
					name = adv_dip_backed_by
					value = root
				}
				add_momentum = momentum_large
			}
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 50 }
			# The money moves in silence: no relations hit, no marker on the
			# target - unless the interference is discovered.
			root = { save_scope_as = spymaster }
			random_list = {
				85 = {}
				15 = {
					scope:target = {
						trigger_event = {
							id = adv_dip_event.225
							days = 14
						}
					}
				}
			}
		}
	}
}

# Promotes an operative to Director of Intelligence. The post must be vacant -
# to replace a sitting Director you must fire them first, with everything that
# entails.
adv_dip_sgui_appoint_director = {
	scope = country
	saved_scopes = { candidate }

	is_shown = {
		always = yes
	}

	is_valid = {
		scope:candidate ?= {
			is_busy = no
			has_role = character_role_operative
			NOT = { has_variable = adv_dip_captured }
		}
		custom_tooltip = {
			text = adv_dip_director_post_filled_tt
			NOT = {
				any_scope_character = {
					has_role = character_role_spymaster
				}
			}
		}
	}

	effect = {
		custom_tooltip = adv_dip_appoint_director_tt
		if = {
			limit = { exists = scope:candidate }
			remove_list_variable = {
				name = adv_dip_operatives
				target = scope:candidate
			}
			add_to_variable_list = {
				name = adv_dip_director
				target = scope:candidate
			}
			scope:candidate = {
				remove_character_role = character_role_operative
				add_character_role = character_role_spymaster
			}
		}
	}
}

# Puts the chosen exile on the sealed train. Root = us; scope:rev = the exile.
adv_dip_sgui_smuggle_pick = {
	scope = country
	saved_scopes = { rev }

	is_shown = {
		always = yes
	}

	is_valid = {
		has_variable = adv_dip_smuggle_pending
		scope:rev ?= {
			is_in_exile_pool = yes
		}
	}

	effect = {
		custom_tooltip = adv_dip_smuggle_pick_tt
		if = {
			limit = {
				exists = scope:rev
				has_variable = adv_dip_smuggle_pending
			}
			var:adv_dip_smuggle_pending ?= {
				save_scope_as = spy_target_country
			}
			scope:rev = {
				save_scope_as = smuggled_revolutionary
				transfer_character = scope:spy_target_country
			}
			remove_variable = adv_dip_smuggle_pending
			clear_variable_list = adv_dip_smuggle_candidates
			trigger_event = { id = adv_dip_event.170 }
		}
	}
}
