remove_assassination_events_effect = {
	adv_dip_release_operative = yes
	remove_variable = active_assassination
	remove_variable = ongoing_plans

	# The modifier is timed (25 months) and absent on abort paths and long plots.

	# The victim lives in the target country, not the initiator's scope; clean the
	# tag there before the country variable itself is removed.
	var:assassination_target_country ?= {
		every_scope_character = {
			limit = { has_variable = assassination_target_var }
			remove_variable = assassination_target_var
		}
	}
	remove_variable = assassination_target_country
}

# Releases the operative used by an assassination plot. On exposure the operative
# may be captured (held by the target) or killed outright.
adv_dip_release_operative = {
	every_scope_character = {
		limit = { has_variable = assassination_leader_var }
		remove_variable = assassination_leader_var
		set_character_busy_and_immortal = no
	}
}

# Captured operatives stay in the adv_dip_operatives roster (the console shows them
# as Captured); executed ones are removed by the monthly dead-operative prune.
adv_dip_lose_operative = {
	every_scope_character = {
		limit = { has_variable = assassination_leader_var }
		random_list = {
			50 = {
				custom_tooltip = adv_dip_operative_captured_tt
				remove_variable = assassination_leader_var
				set_character_busy_and_immortal = no
				# Both timers match: if no ransom treaty frees them first, the
				# operative escapes on their own after five years.
				set_variable = {
					name = adv_dip_captured
					months = 60
				}
				if = {
					limit = { exists = scope:spy_target_country }
					set_variable = {
						name = adv_dip_captor
						value = scope:spy_target_country
						months = 60
					}
				}
			}
			50 = {
				custom_tooltip = adv_dip_operative_executed_tt
				remove_variable = assassination_leader_var
				set_character_busy_and_immortal = no
				kill_character = yes
			}
		}
	}
}
