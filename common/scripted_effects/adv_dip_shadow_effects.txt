shadow_war_score_minor_effect = {
	if = {
		limit = { NOT = { has_variable = shadow_war_score } }
		set_variable = {
			name = shadow_war_score
			value = 0
		}
	}
	change_variable = {
		name = shadow_war_score
		add = 1
	}
	shadow_war_leadership_check_effect = yes
}

shadow_war_score_major_effect = {
	if = {
		limit = { NOT = { has_variable = shadow_war_score } }
		set_variable = {
			name = shadow_war_score
			value = 0
		}
	}
	change_variable = {
		name = shadow_war_score
		add = 3
	}
	shadow_war_leadership_check_effect = yes
}

shadow_war_leadership_check_effect = {
	if = {
		limit = {
			exists = global_var:shadow_war_leader
			global_var:shadow_war_leader = { is_country_alive = no }
		}
		remove_global_variable = shadow_war_leader
	}
	if = {
		limit = {
			OR = {
				has_law = law_type:law_consular_intelligence
				has_law = law_type:law_spy_agency
				has_law = law_type:law_directorate_of_state_security
				has_law = law_type:law_deniable_assets
			}
			has_variable = shadow_war_score
			OR = {
				NOT = { exists = global_var:shadow_war_leader }
				AND = {
					NOT = { global_var:shadow_war_leader ?= this }
					global_var:shadow_war_leader = {
						has_variable = shadow_war_score
					}
					var:shadow_war_score > global_var:shadow_war_leader.var:shadow_war_score
				}
			}
		}
		if = {
			limit = { exists = global_var:shadow_war_leader }
			global_var:shadow_war_leader = {
				remove_modifier = master_of_the_shadow_war_modifier
				trigger_event = { id = adv_dip_event.166 }
			}
		}
		set_global_variable = {
			name = shadow_war_leader
			value = this
		}
		add_modifier = {
			name = master_of_the_shadow_war_modifier
		}
		trigger_event = { id = adv_dip_event.165 }
	}
}
