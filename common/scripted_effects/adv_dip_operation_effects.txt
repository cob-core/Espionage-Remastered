# Core secret-operation machinery (v5.0). Operations live in per-type variable
# lists on the initiator; pacts are gone.

# Espionage money never moves through modifiers: a country modifier is public
# and would advertise the spending to every observer (the old "Building an
# Intelligence Network" tell). COST names a script value; WEEKS sizes the lump
# in weeks of that value (roughly half the old modifiers' nominal totals,
# matching what their replace-semantics actually collected).
adv_dip_covert_charge = {
	add_treasury = {
		value = $COST$
		multiply = -$WEEKS$
	}
}

adv_dip_start_operation = {
	add_to_variable_list = {
		name = adv_dip_op_$ACTION$
		target = scope:target_country
	}
}

# Queues an operation behind operative selection. CODE is the numeric action id
# understood by adv_dip_dispatch_pending_operation. The player picks an operative
# in adv_dip_event.200; the AI assigns one at random and launches immediately.
# Root = initiator, scope:target_country = the lens target.
adv_dip_begin_operation = {
	scope:target_country = {
		save_scope_as = spy_target_country
	}
	# Timed as a backstop: if the selection event is ever lost, the launch gate
	# must not stay blocked forever.
	set_variable = {
		name = adv_dip_pending_op
		value = $CODE$
		months = 2
	}
	set_variable = {
		name = adv_dip_pending_target
		value = scope:target_country
		months = 2
	}
	if = {
		limit = { is_player = yes }
		trigger_event = { id = adv_dip_event.200 popup = yes }
	}
	else = {
		random_scope_character = {
			limit = {
				has_role = character_role_operative
				is_busy = no
				NOT = { has_variable = adv_dip_captured }
			}
			save_scope_as = chosen_operative
		}
		adv_dip_dispatch_pending_operation = yes
	}
}

# Binds scope:chosen_operative to the pending operation and launches it: list
# operations start their variable-list entry, one-shots fire their event chain,
# and the launch cost is charged here — so an aborted selection costs nothing.
# Clears the pending state on every path.
adv_dip_dispatch_pending_operation = {
	if = {
		limit = { NOT = { exists = scope:spy_target_country } }
		var:adv_dip_pending_target ?= { save_scope_as = spy_target_country }
	}
	if = {
		limit = {
			exists = scope:chosen_operative
			exists = scope:spy_target_country
			has_variable = adv_dip_pending_op
		}
		if = {
			limit = { var:adv_dip_pending_op = 1 }
			adv_dip_start_operation = { ACTION = sow_discontent }
			adv_dip_assign_operative = { ACTION = sow_discontent }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 26 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 2 }
			adv_dip_start_operation = { ACTION = support_nationalists }
			adv_dip_assign_operative = { ACTION = support_nationalists }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 26 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 3 }
			adv_dip_start_operation = { ACTION = agitate_for_liberty }
			adv_dip_assign_operative = { ACTION = agitate_for_liberty }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 26 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 4 }
			adv_dip_start_operation = { ACTION = steal_technology }
			adv_dip_assign_operative = { ACTION = steal_technology }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_small WEEKS = 26 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 5 }
			adv_dip_start_operation = { ACTION = sabotage_recruitment }
			adv_dip_assign_operative = { ACTION = sabotage_recruitment }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_small WEEKS = 26 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 6 }
			adv_dip_start_operation = { ACTION = election_interference }
			adv_dip_assign_operative = { ACTION = election_interference }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 26 }
			# Campaign already running: the ballot opens immediately instead of
			# waiting for the monthly pulse.
			if = {
				limit = {
					is_player = yes
					scope:spy_target_country = { in_election_campaign = yes }
					NOT = {
						is_target_in_variable_list = {
							name = adv_dip_election_choice
							target = scope:spy_target_country
						}
					}
				}
				add_to_variable_list = {
					name = adv_dip_election_choice
					target = scope:spy_target_country
				}
				trigger_event = { id = adv_dip_event.81 popup = yes }
			}
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 7 }
			adv_dip_start_operation = { ACTION = steal_commander }
			adv_dip_assign_operative = { ACTION = steal_commander }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_small WEEKS = 26 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 8 }
			adv_dip_start_operation = { ACTION = clandestine_arms_pipeline }
			adv_dip_assign_operative = { ACTION = clandestine_arms_pipeline }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 26 }
			scope:spy_target_country = {
				add_modifier = {
					name = adv_dip_arms_pipeline_target_modifier
				}
			}
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 9 }
			adv_dip_start_operation = { ACTION = fund_black_market_slavery }
			adv_dip_assign_operative = { ACTION = fund_black_market_slavery }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 26 }
			# The contraband flows to us: the import boost is ours.
			add_modifier = {
				name = adv_dip_black_market_modifier
			}
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 10 }
			adv_dip_start_operation = { ACTION = support_slaves }
			adv_dip_assign_operative = { ACTION = support_slaves }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_small WEEKS = 26 }
			scope:spy_target_country = {
				add_modifier = {
					name = adv_dip_support_slaves_target_modifier
				}
			}
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 11 }
			scope:spy_target_country = {
				save_scope_as = naval_intel_target
			}
			set_variable = {
				name = naval_intel_target_var
				value = scope:naval_intel_target
			}
			adv_dip_assign_operative = { ACTION = naval_intelligence }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 26 }
			trigger_event = { id = adv_dip_event.140 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 12 }
			set_variable = {
				name = recently_sealed_train
				months = 60
			}
			adv_dip_assign_operative = { ACTION = sealed_train }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_very_large WEEKS = 50 }
			if = {
				limit = { is_player = yes }
				# The iterator body runs in the EXILE's scope, and character
				# creation is deferred - so every list write must be root-wrapped
				# and live inside on_created (the recruit-button pattern).
				every_character_in_exile_pool = {
					root = {
						add_to_variable_list = {
							name = adv_dip_smuggle_candidates
							target = prev
						}
					}
				}
				# An empty exile pool is no obstacle: the service finds its own
				# firebrands among the target's malcontents, exiled to order.
				if = {
					limit = { NOT = { any_character_in_exile_pool = { always = yes } } }
					scope:spy_target_country = {
						create_character = {
							culture = primary_culture
							ideology = ideology_radical
							on_created = {
								exile_character = yes
								root = {
									add_to_variable_list = {
										name = adv_dip_smuggle_candidates
										target = prev
									}
								}
							}
						}
						create_character = {
							culture = primary_culture
							ideology = ideology_communist
							on_created = {
								exile_character = yes
								root = {
									add_to_variable_list = {
										name = adv_dip_smuggle_candidates
										target = prev
									}
								}
							}
						}
						create_character = {
							culture = primary_culture
							ideology = ideology_anarchist
							on_created = {
								exile_character = yes
								root = {
									add_to_variable_list = {
										name = adv_dip_smuggle_candidates
										target = prev
									}
								}
							}
						}
					}
				}
				set_variable = {
					name = adv_dip_smuggle_pending
					value = scope:spy_target_country
					months = 6
				}
				trigger_event = { id = adv_dip_event.230 popup = yes }
			}
			else = {
				trigger_event = { id = adv_dip_event.170 }
			}
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 13 }
			scope:spy_target_country = {
				save_scope_as = development_target
			}
			adv_dip_assign_operative = { ACTION = develop_nation }
			# The development program runs ~34 months (100 progress at 3/month);
			# the charge covers the whole window.
			adv_dip_covert_charge = { COST = spy_cost_multiplier_medium WEEKS = 68 }
			trigger_event = { id = adv_dip_event.110 }
		}
		else_if = {
			limit = { var:adv_dip_pending_op = 14 }
			set_variable = {
				name = false_flag_patsy
				value = scope:spy_target_country
				months = 60
			}
			adv_dip_assign_operative = { ACTION = false_flag }
			adv_dip_covert_charge = { COST = spy_cost_multiplier_large WEEKS = 120 }
		}
	}
	remove_variable = adv_dip_pending_op
	remove_variable = adv_dip_pending_target
}

# Marks scope:chosen_operative as running one operation. The busy flag blocks
# double-assignment; immortality mirrors the assassination chain so a mission
# cannot lose its agent to random death mid-story. adv_dip_settling_in shields
# fresh assignments from the monthly reconciler until their chain has set up
# its own state (e.g. ongoing_naval_intel).
adv_dip_assign_operative = {
	scope:chosen_operative ?= {
		set_character_busy_and_immortal = yes
		set_variable = adv_dip_assigned_$ACTION$
		set_variable = {
			name = adv_dip_assignment_target
			value = scope:spy_target_country
		}
		set_variable = {
			name = adv_dip_settling_in
			months = 3
		}
	}
	# Parallel agent list drives the console's operation rows.
	if = {
		limit = { exists = scope:chosen_operative }
		add_to_variable_list = {
			name = adv_dip_agents_$ACTION$
			target = scope:chosen_operative
		}
	}
}

# Frees an operative for reassignment. Character scope.
adv_dip_free_operative = {
	set_character_busy_and_immortal = no
	remove_variable = adv_dip_assigned_$ACTION$
	remove_variable = adv_dip_assignment_target
	remove_variable = adv_dip_settling_in
	owner ?= {
		if = {
			limit = { has_variable_list = adv_dip_agents_$ACTION$ }
			remove_list_variable = {
				name = adv_dip_agents_$ACTION$
				target = prev
			}
		}
	}
}

# Stops one operation. TARGET must be a saved scope name (without scope: prefix).
# Defensive by design: every caller may pass a scope that no longer resolves
# (a dead country, a stale UI row), so the whole body is existence-guarded.
adv_dip_stop_operation = {
	if = {
		limit = { exists = scope:$TARGET$ }

		remove_list_variable = {
			name = adv_dip_op_$ACTION$
			target = scope:$TARGET$
		}

		every_scope_character = {
			limit = {
				has_variable = adv_dip_assigned_$ACTION$
				var:adv_dip_assignment_target ?= scope:$TARGET$
			}
			adv_dip_free_operative = { ACTION = $ACTION$ }
		}

		scope:$TARGET$ = {
			if = {
				limit = { has_modifier = adv_dip_arms_pipeline_target_modifier }
				remove_modifier = adv_dip_arms_pipeline_target_modifier
			}
			if = {
				limit = { has_modifier = adv_dip_support_slaves_target_modifier }
				remove_modifier = adv_dip_support_slaves_target_modifier
			}
		}

		# Our import boost outlives any single route; it ends with the last one.
		if = {
			limit = {
				has_modifier = adv_dip_black_market_modifier
				NOT = {
					any_country = {
						root = {
							is_target_in_variable_list = {
								name = adv_dip_op_fund_black_market_slavery
								target = prev
							}
						}
					}
				}
			}
			remove_modifier = adv_dip_black_market_modifier
		}
	}
}

# Character scope. Frees an operative whose list operation no longer exists —
# the catch-all release for wind-downs, target annexation and event-chain ends,
# so no resolution path ever has to remember to release its agent by hand.
adv_dip_release_if_orphaned = {
	if = {
		limit = {
			has_variable = adv_dip_assigned_$ACTION$
			NOT = { has_variable = adv_dip_settling_in }
		}
		# Sentinel first: saved scopes persist across loop iterations, so if the
		# target var no longer resolves the previous operative's target would
		# leak into this check. Root is never a valid operation target, so an
		# unresolvable var reads as "orphaned" — which it is.
		owner = { save_scope_as = adv_dip_check_target }
		var:adv_dip_assignment_target ?= { save_scope_as = adv_dip_check_target }
		if = {
			limit = {
				owner = {
					NOT = {
						is_target_in_variable_list = {
							name = adv_dip_op_$ACTION$
							target = scope:adv_dip_check_target
						}
					}
				}
			}
			adv_dip_free_operative = { ACTION = $ACTION$ }
		}
		else_if = {
			# Self-healing: a live assignment must appear in the console's
			# agent list (also backfills saves from before the list existed).
			limit = {
				owner = {
					NOT = {
						is_target_in_variable_list = {
							name = adv_dip_agents_$ACTION$
							target = prev
						}
					}
				}
			}
			owner = {
				add_to_variable_list = {
					name = adv_dip_agents_$ACTION$
					target = prev
				}
			}
		}
	}
}

# Character scope. Frees a one-shot mission operative once the owner no longer
# carries the mission's tracking variable ($FLAG$).
adv_dip_release_if_mission_over = {
	if = {
		limit = {
			has_variable = adv_dip_assigned_$ACTION$
			NOT = { has_variable = adv_dip_settling_in }
			owner = { NOT = { has_variable = $FLAG$ } }
		}
		adv_dip_free_operative = { ACTION = $ACTION$ }
	}
}

# Country scope, monthly. Reconciles every busy operative against the live
# operation state and releases the orphans.
adv_dip_reconcile_assignments = {
	if = {
		limit = { has_variable_list = adv_dip_operatives }
		every_in_list = {
			variable = adv_dip_operatives
			limit = { is_busy = yes }
			adv_dip_release_if_orphaned = { ACTION = sow_discontent }
			adv_dip_release_if_orphaned = { ACTION = support_nationalists }
			adv_dip_release_if_orphaned = { ACTION = agitate_for_liberty }
			adv_dip_release_if_orphaned = { ACTION = steal_technology }
			adv_dip_release_if_orphaned = { ACTION = sabotage_recruitment }
			adv_dip_release_if_orphaned = { ACTION = election_interference }
			adv_dip_release_if_orphaned = { ACTION = steal_commander }
			adv_dip_release_if_orphaned = { ACTION = clandestine_arms_pipeline }
			adv_dip_release_if_orphaned = { ACTION = fund_black_market_slavery }
			adv_dip_release_if_orphaned = { ACTION = support_slaves }
			adv_dip_release_if_mission_over = { ACTION = naval_intelligence FLAG = ongoing_naval_intel }
			adv_dip_release_if_mission_over = { ACTION = develop_nation FLAG = ongoing_development }
			adv_dip_release_if_mission_over = { ACTION = false_flag FLAG = false_flag_patsy }
			# The sealed-train run has no ongoing tracker: the courier returns
			# once the settling-in window lapses (~3 months in the field).
			if = {
				limit = {
					has_variable = adv_dip_assigned_sealed_train
					NOT = { has_variable = adv_dip_settling_in }
				}
				adv_dip_free_operative = { ACTION = sealed_train }
			}
		}
	}
}

# Monthly die for one target of one operation type. Called from the monthly
# loop in the TARGET's scope (spy_target_country saved), so every own-country
# read goes through root. CHANCE is the base monthly success percentage; EVENT
# the X1 event id; ACTION the assignment flag; TRAIT the tradecraft trait that
# sharpens the roll. The assigned operative's experience tier compounds on top.
adv_dip_operation_die = {
	random = {
		chance = {
			value = $CHANCE$
			if = {
				limit = { root = { institution_investment_level:institution_espionage >= spy_target_espionage_plus_1 } }
				multiply = 1.5
			}
			else_if = {
				limit = { root = { institution_investment_level:institution_espionage <= spy_target_espionage_minus_1 } }
				multiply = 0.65
			}
			if = {
				limit = {
					root = {
						any_scope_character = {
							has_variable = adv_dip_assigned_$ACTION$
							var:adv_dip_assignment_target ?= scope:spy_target_country
							has_trait = $TRAIT$
						}
					}
				}
				multiply = 1.3
			}
			if = {
				limit = {
