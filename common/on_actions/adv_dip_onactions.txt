
# Root = Country
# scope:target = Uprising country
on_secession_start = {
	on_actions = {
		adv_dip_secession_hook
	}
}

adv_dip_secession_hook = {
	effect = {
		if = {
			limit = {
				any_country = {
					is_target_in_variable_list = { name = adv_dip_op_support_nationalists target = root }
				}
			}
			every_country = {
				limit = {
					is_target_in_variable_list = { name = adv_dip_op_support_nationalists target = root }
				}
				change_relations = {
					country = scope:target
					value = 200
				}
				set_owes_obligation_to = {
					country = scope:target
					setting = no
				}
			}
		}
		if = {
			limit = {
				any_country = {
					any_scope_culture ={
						is_primary_culture_of = prev
						is_primary_culture_of = scope:target
					}
				}
			}
			every_country = {
				limit = {
					any_scope_culture ={
						is_primary_culture_of = prev
						is_primary_culture_of = scope:target
					}
				}
				change_relations = {
					country = scope:target
					value = 200
				}
				set_owes_obligation_to = {
					country = scope:target
					setting = no
				}
			}
		}
	}
}

on_yearly_pulse_country = {
	on_actions = {
		adv_dip_espionage_yearly
		adv_dip_turned_network_review
	}
}

adv_dip_espionage_yearly = {
	effect = {
		# AI operative management: the console is player-only, so the AI recruits
		# and appoints through script or it could never run character-led operations.
		if = {
			limit = {
				is_player = no
				OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
				scaled_debt <= 0
				NOT = {
					list_size = {
						name = adv_dip_operatives
						value >= institution_investment_level:institution_espionage
					}
				}
			}
			create_character = {
				role = character_role_operative
				culture = primary_culture
				on_created = {
					save_scope_as = adv_dip_ai_operative
					random_list = {
						1 = { add_trait = cryptographer }
						1 = { add_trait = master_of_disguise }
						1 = { add_trait = saboteur }
						1 = { add_trait = silver_tongue }
					}
				}
			}
			add_to_variable_list = {
				name = adv_dip_operatives
				target = scope:adv_dip_ai_operative
			}
			# Second recruit in the same year while still under cap: operations
			# bind operatives for their whole lifetime, so one hire a year would
			# starve the AI's espionage program within a decade.
			if = {
				limit = {
					NOT = {
						list_size = {
							name = adv_dip_operatives
							value >= institution_investment_level:institution_espionage
						}
					}
				}
				create_character = {
					role = character_role_operative
					culture = primary_culture
					on_created = {
						save_scope_as = adv_dip_ai_operative_2
						random_list = {
							1 = { add_trait = cryptographer }
							1 = { add_trait = master_of_disguise }
							1 = { add_trait = saboteur }
							1 = { add_trait = silver_tongue }
						}
					}
				}
				add_to_variable_list = {
					name = adv_dip_operatives
					target = scope:adv_dip_ai_operative_2
				}
			}
		}
		if = {
			limit = {
				is_player = no
				NOT = { has_variable_list = adv_dip_director }
				has_variable_list = adv_dip_operatives
			}
			random_in_list = {
				variable = adv_dip_operatives
				save_scope_as = adv_dip_new_director
				remove_character_role = character_role_operative
				add_character_role = character_role_spymaster
			}
			remove_list_variable = {
				name = adv_dip_operatives
				target = scope:adv_dip_new_director
			}
			add_to_variable_list = {
				name = adv_dip_director
				target = scope:adv_dip_new_director
			}
		}
		if = {
			limit = {
				OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
				NOT = { has_journal_entry = adv_dip_je_shadow_war }
			}
			add_journal_entry = {
				type = adv_dip_je_shadow_war
			}
		}
		# Underpaid spies shop their knowledge around: at low government wages
		# there is a yearly chance an operative approaches a hostile power.
		if = {
			limit = {
				government_wage_level <= low
				any_scope_character = {
					has_role = character_role_operative
					is_busy = no
					NOT = { has_variable = adv_dip_captured }
				}
				any_country = {
					NOT = { this = root }
					OR = {
						has_diplomatic_pact = {
							who = root
							type = rivalry
						}
						relations:root <= relations_threshold:hostile
					}
				}
			}
			# random_list weight modifiers are banned in this codebase (silent
			# 1.13 failures) - tiered rolls instead.
			if = {
				limit = { government_wage_level = very_low }
				random_list = {
					88 = {}
					12 = {
						adv_dip_wage_defection = yes
					}
				}
			}
			else = {
				random_list = {
					94 = {}
					6 = {
						adv_dip_wage_defection = yes
					}
				}
			}
		}
		# A captured operative in foreign cells is a standing liability: each
		# year they risk talking. Ransoming them back (Operative Repatriation
		# treaty article) or disavowing them are the only ways to end it early.
		if = {
			limit = {
				any_scope_character = {
					has_role = character_role_operative
					has_variable = adv_dip_captured
					has_variable = adv_dip_captor
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_operative
					has_variable = adv_dip_captured
					has_variable = adv_dip_captor
				}
				save_scope_as = adv_dip_interrogated
				var:adv_dip_captor ?= {
					save_scope_as = adv_dip_captor_country
				}
			}
			if = {
				limit = {
					exists = scope:adv_dip_interrogated
					exists = scope:adv_dip_captor_country
				}
				trigger_event = { id = adv_dip_event.210 }
			}
		}
		# Great Game score decay
		if = {
			limit = { has_variable = shadow_war_score }
			change_variable = {
				name = shadow_war_score
				multiply = 0.9
			}
		}
		# Kompromat dossier accrual while any espionage network runs. The list
		# check must run in OUR scope (the lists live on the initiator) — in the
		# member's scope it would ask whether they spy on us.
		if = {
			limit = {
				OR = {
					has_variable = ongoing_naval_intel
					any_country = {
						root = {
							OR = {
								is_target_in_variable_list = { name = adv_dip_op_sow_discontent target = prev }
								is_target_in_variable_list = { name = adv_dip_op_support_nationalists target = prev }
								is_target_in_variable_list = { name = adv_dip_op_agitate_for_liberty target = prev }
								is_target_in_variable_list = { name = adv_dip_op_steal_technology target = prev }
								is_target_in_variable_list = { name = adv_dip_op_election_interference target = prev }
								is_target_in_variable_list = { name = adv_dip_op_support_slaves target = prev }
								is_target_in_variable_list = { name = adv_dip_op_sabotage_recruitment target = prev }
								is_target_in_variable_list = { name = adv_dip_op_steal_commander target = prev }
							}
						}
					}
				}
			}
			if = {
				limit = { NOT = { has_variable = kompromat_dossier_progress } }
				set_variable = {
					name = kompromat_dossier_progress
					value = 0
				}
			}
			change_variable = {
				name = kompromat_dossier_progress
				add = 1
			}
			if = {
				limit = {
					var:kompromat_dossier_progress >= 5
				}
				remove_variable = kompromat_dossier_progress
				trigger_event = { id = adv_dip_event.180 }
			}
		}

	}
}

adv_dip_turned_network_review = {
	effect = {
		if = {
			limit = {
				any_country = {
					OR = {
						var:turned_steal_technology_by ?= prev
						var:turned_election_interference_by ?= prev
						var:turned_sow_discontent_by ?= prev
					}
				}
			}
			trigger_event = { id = adv_dip_event.160 }
		}
	}
}

adv_dip_espionage_monthly = {
	effect = {
		set_variable = {
			name = adv_dip_upkeep_total
			value = 0
		}
		# Prune a dead or departed Director before granting their bonus
		if = {
			limit = { has_variable_list = adv_dip_director }
			every_in_list = {
				variable = adv_dip_director
				limit = {
					OR = {
						is_character_alive = no
						NOT = { owner ?= root }
					}
				}
				add_to_local_variable_list = {
					name = adv_dip_prune_director
					target = this
				}
			}
			every_in_local_list = {
				variable = adv_dip_prune_director
				save_scope_as = adv_dip_lost_director
				root = {
					remove_list_variable = {
						name = adv_dip_director
						target = scope:adv_dip_lost_director
					}
				}
			}
		}
		# Director of Intelligence bonus, maintained while one is seated
		if = {
			limit = { has_variable_list = adv_dip_director }
			add_modifier = {
				name = adv_dip_director_modifier
				months = 2
			}
		}
		# Prune operatives who have died or left our service
		if = {
			limit = { has_variable_list = adv_dip_operatives }
			every_in_list = {
				variable = adv_dip_operatives
				limit = {
					OR = {
						is_character_alive = no
						NOT = { owner ?= root }
					}
				}
				add_to_local_variable_list = {
					name = adv_dip_prune_operative
					target = this
				}
			}
			every_in_local_list = {
				variable = adv_dip_prune_operative
				save_scope_as = adv_dip_lost_operative
				root = {
					remove_list_variable = {
						name = adv_dip_operatives
						target = scope:adv_dip_lost_operative
					}
				}
			}
		}
		# Free any operative whose operation has ended by a path that could not
		# release them directly (annexation, chain resolution, expiry).
		adv_dip_reconcile_assignments = yes
		# An unanswered smuggle slate expires with its pending marker.
		if = {
			limit = {
				has_variable_list = adv_dip_smuggle_candidates
				NOT = { has_variable = adv_dip_smuggle_pending }
			}
			clear_variable_list = adv_dip_smuggle_candidates
		}
		# A pending election-interference choice dies with the campaign.
		if = {
			limit = { has_variable_list = adv_dip_election_choice }
			every_in_list = {
				variable = adv_dip_election_choice
				limit = { in_election_campaign = no }
				add_to_local_variable_list = {
					name = adv_dip_stale_elections
					target = this
				}
			}
			every_in_local_list = {
				variable = adv_dip_stale_elections
				save_scope_as = adv_dip_stale_election
				root = {
					remove_list_variable = {
						name = adv_dip_election_choice
						target = scope:adv_dip_stale_election
					}
				}
			}
		}
		# Experience: months on assignment accrue toward tier promotions, and
		# every operative not in enemy hands draws a wage scaled by their tier.
		if = {
			limit = { has_variable_list = adv_dip_operatives }
			every_in_list = {
				variable = adv_dip_operatives
				if = {
					limit = {
						is_busy = yes
						NOT = { has_variable = adv_dip_captured }
					}
					if = {
						limit = { NOT = { has_variable = adv_dip_mission_months } }
						set_variable = {
							name = adv_dip_mission_months
							value = 0
						}
					}
					change_variable = {
						name = adv_dip_mission_months
						add = 1
					}
					if = {
						limit = {
							var:adv_dip_mission_months >= 72
							NOT = { has_trait = adv_dip_master_spy }
						}
						if = {
							limit = { has_trait = adv_dip_veteran }
							remove_trait = adv_dip_veteran
						}
						add_trait = adv_dip_master_spy
					}
					else_if = {
						limit = {
							var:adv_dip_mission_months >= 36
							NOR = {
								has_trait = adv_dip_veteran
								has_trait = adv_dip_master_spy
							}
						}
						if = {
							limit = { has_trait = adv_dip_seasoned }
							remove_trait = adv_dip_seasoned
						}
						add_trait = adv_dip_veteran
					}
					else_if = {
						limit = {
							var:adv_dip_mission_months >= 12
							NOR = {
								has_trait = adv_dip_seasoned
								has_trait = adv_dip_veteran
								has_trait = adv_dip_master_spy
							}
						}
						add_trait = adv_dip_seasoned
					}
				}
				if = {
					limit = { NOT = { has_variable = adv_dip_captured } }
					if = {
						limit = { has_trait = adv_dip_master_spy }
						root = {
							change_variable = {
								name = adv_dip_upkeep_total
								add = adv_dip_wage_master
							}
						}
					}
					else_if = {
						limit = { has_trait = adv_dip_veteran }
						root = {
							change_variable = {
								name = adv_dip_upkeep_total
								add = adv_dip_wage_veteran
							}
						}
					}
					else_if = {
						limit = { has_trait = adv_dip_seasoned }
						root = {
							change_variable = {
								name = adv_dip_upkeep_total
								add = adv_dip_wage_seasoned
							}
						}
					}
					else = {
						root = {
							change_variable = {
								name = adv_dip_upkeep_total
								add = adv_dip_wage_base
							}
						}
					}
				}
			}
		}
		# The Director of Intelligence draws an executive salary
		if = {
			limit = { has_variable_list = adv_dip_director }
			change_variable = {
				name = adv_dip_upkeep_total
				add = adv_dip_wage_director
			}
		}
		# AI solvency abort: a broke AI winds down its networks
		if = {
			limit = {
				is_player = no
				OR = {
					scaled_debt > 0.25
					in_default = yes
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_sow_discontent }
				every_in_list = {
					variable = adv_dip_op_sow_discontent
					add_to_local_variable_list = {
						name = adv_dip_winddown_sow_discontent
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_sow_discontent
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = sow_discontent TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_support_nationalists }
				every_in_list = {
					variable = adv_dip_op_support_nationalists
					add_to_local_variable_list = {
						name = adv_dip_winddown_support_nationalists
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_support_nationalists
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = support_nationalists TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_agitate_for_liberty }
				every_in_list = {
					variable = adv_dip_op_agitate_for_liberty
					add_to_local_variable_list = {
						name = adv_dip_winddown_agitate_for_liberty
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_agitate_for_liberty
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = agitate_for_liberty TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_steal_technology }
				every_in_list = {
					variable = adv_dip_op_steal_technology
					add_to_local_variable_list = {
						name = adv_dip_winddown_steal_technology
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_steal_technology
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = steal_technology TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_sabotage_recruitment }
				every_in_list = {
					variable = adv_dip_op_sabotage_recruitment
					add_to_local_variable_list = {
						name = adv_dip_winddown_sabotage_recruitment
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_sabotage_recruitment
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = sabotage_recruitment TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_election_interference }
				every_in_list = {
					variable = adv_dip_op_election_interference
					add_to_local_variable_list = {
						name = adv_dip_winddown_election_interference
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_election_interference
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = election_interference TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_steal_commander }
				every_in_list = {
					variable = adv_dip_op_steal_commander
					add_to_local_variable_list = {
						name = adv_dip_winddown_steal_commander
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_steal_commander
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = steal_commander TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_clandestine_arms_pipeline }
				every_in_list = {
					variable = adv_dip_op_clandestine_arms_pipeline
					add_to_local_variable_list = {
						name = adv_dip_winddown_clandestine_arms_pipeline
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_clandestine_arms_pipeline
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = clandestine_arms_pipeline TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_fund_black_market_slavery }
				every_in_list = {
					variable = adv_dip_op_fund_black_market_slavery
					add_to_local_variable_list = {
						name = adv_dip_winddown_fund_black_market_slavery
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_fund_black_market_slavery
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = fund_black_market_slavery TARGET = adv_dip_wound_down }
					}
				}
			}
			if = {
				limit = { has_variable_list = adv_dip_op_support_slaves }
				every_in_list = {
					variable = adv_dip_op_support_slaves
					add_to_local_variable_list = {
						name = adv_dip_winddown_support_slaves
						target = this
					}
				}
				every_in_local_list = {
					variable = adv_dip_winddown_support_slaves
					save_scope_as = adv_dip_wound_down
					root = {
						adv_dip_stop_operation = { ACTION = support_slaves TARGET = adv_dip_wound_down }
					}
				}
			}
		}
		# --- sow_discontent ---
		if = {
			limit = { has_variable_list = adv_dip_op_sow_discontent }
			every_in_list = {
				variable = adv_dip_op_sow_discontent
				save_scope_as = spy_target_country
				if = {
					limit = {
						OR = {
							scope:spy_target_country = { is_country_type = decentralized }
						root = { has_war_with = scope:spy_target_country }
						}
					}
					add_to_local_variable_list = {
						name = adv_dip_stop_sow_discontent
						target = scope:spy_target_country
					}
				}
				else = {
