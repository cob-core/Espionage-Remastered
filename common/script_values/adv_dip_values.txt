# Cost of the counter-piracy journal-entry buttons: 0.015% of own GDP, rounded
# to nearest 100. Runs on the piracy victim, so it scales off root, not a spy scope.
piracy_cost_multiplier = {
	value = root.gdp
	divide = 100000
	multiply = 0.15
	round = yes
	multiply = 100
}

piracy_initiator = {
    value = scope:piracy_initiator.naval_power_projection
    add = 1
}

piracy_target = {
    value = scope:piracy_target.naval_power_projection
    add = 1
}

piracy_naval_difference = {
    value = piracy_initiator
    divide = piracy_target
}

piracy_naval_modifier = {
    value = 1
    # Cap the naval ratio effect between 0.5x and 1.5x
    # If initiator has more navy, add up to +50%
    if = {
        limit = {
            piracy_naval_difference > 1
        }
        add = 0.5
        # Subtract excess if ratio is too high (e.g., if ratio is 3, we added 0.5 but want max +0.5)
        if = {
            limit = {
                piracy_naval_difference > 2
            }
            subtract = piracy_naval_difference
            add = 2
        }
    }
    # If target has more navy, subtract up to -50%
    else_if = {
        limit = {
            piracy_naval_difference < 1
        }
        multiply = piracy_naval_difference
        # Ensure it doesn't go below 0.5
        if = {
            limit = {
                piracy_naval_difference < 0.5
            }
            divide = piracy_naval_difference
            multiply = 0.5
        }
    }
}
