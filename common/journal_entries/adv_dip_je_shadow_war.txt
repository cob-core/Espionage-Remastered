adv_dip_je_shadow_war = {

	group = je_group_espionage

	widget = {
		gui = "gui/journal_entry_widgets/adv_dip_console_widgets.gui"
		name = "widget_adv_dip_console"
		container = "custom_widget_container_1"
	}

	weight = 100
	transferable = no
	# This is a context = country JE with no involved parties, so the
	# _uninvolved_or_context variant is the one that actually pins it.
	should_be_pinned_by_default_uninvolved_or_context = yes

	is_shown_when_inactive = {
		OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
	}

	possible = {
		OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		}
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = shadow_war_score } }
			set_variable = {
				name = shadow_war_score
				value = 0
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = adv_dip_je_shadow_war_leading
				trigger = {
					exists = global_var:shadow_war_leader
					global_var:shadow_war_leader ?= this
				}
			}
			triggered_desc = {
				desc = adv_dip_je_shadow_war_playing
				trigger = { always = yes }
			}
		}
	}

	invalid = {
		NOT = { OR = {
			has_law = law_type:law_consular_intelligence
			has_law = law_type:law_spy_agency
			has_law = law_type:law_directorate_of_state_security
			has_law = law_type:law_deniable_assets
		} }
	}

	on_invalid = {
	}
}
